```python
#: Registered skills that roll dice INSIDE themselves. A receipt from any of
#: these IS the roll -- counting only roll_dice/resolve_skill_check scored a
#: real work shift narrated as a success at 0.2 and forced a retry, which
#: pushed the narrator away from reporting real outcomes.
ROLLING_SKILLS = frozenset({
    "roll_dice", "resolve_skill_check", "work", "forage",
    "encounter_approach", "resolve_scene_card", "resolve_challenge",
})

_CLAIMS_SUCCESS = re.compile(r"(?i)(?<!not )(?<!n't )\byou (succeed|manage it|pull it off)\b")
_CLAIMS_ARRIVAL = re.compile(r"(?i)\byou (arrive|reach|step into|come out at)\b")


def contradicts(narration: str, receipts: list[dict[str, Any]]) -> str:
    """
    A note when the prose states the OPPOSITE of what the engine decided.

    Deliberately narrow: two unambiguous opposites and nothing else. Success
    over a failed check; arrival over a refused move. Anything subtler is a
    judgement a regex cannot make, and a gate that fires on honest prose is a
    gate somebody deletes.
    """
    for r in receipts:
        result = r.get("result") if isinstance(r.get("result"), dict) else {}
        if r.get("skill") == "resolve_skill_check" and result.get("success") is False:
            if _CLAIMS_SUCCESS.search(narration):
                return "narrated success over a failed check"
        if r.get("skill") == "move_to" and (r.get("refused") or r.get("success") is False):
            if _CLAIMS_ARRIVAL.search(narration):
                return "narrated arrival over a refused move"
    return ""
```

```python
"""
The world moved
===============

A per-turn journal of what changed while the player was not looking, so the
narrator can mention it on the turn it happened and never again.

Every entry is written by the engine system that caused it. The journal is
PRESENTATION, not game state: it is never saved, and it is drained once the
turn's prompt has been built. Nothing reads it to make a decision.
"""
from __future__ import annotations

from typing import Any

KINDS = ("beat", "event", "promise", "band", "standing", "gossip")
MAX_ENTRIES = 8


def note(state: Any, kind: str, text: str, *, location_id: str = "") -> None:
    if kind not in KINDS:
        raise ValueError(f"unknown moved kind {kind!r}")
    text = " ".join(str(text or "").split())
    if not text:
        return
    journal = state.moved
    if any(e["text"] == text for e in journal):
        return
    journal.append({"kind": kind, "text": text, "location_id": location_id})
    del journal[:-MAX_ENTRIES]


def drain(state: Any) -> list[dict[str, Any]]:
    out, state.moved = list(state.moved), []
    return out
```

```python
def moved_block(state: GameState, ledger: Any = None) -> str:
    """What changed since the last turn. Once each; never repeated."""
    rows = [e for e in state.moved if not e.get("location_id") or e["location_id"] == state.location_id]
```
