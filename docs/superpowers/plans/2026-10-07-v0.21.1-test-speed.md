# v0.21.1 Test-Suite Speed-Up Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the default `pytest` run take minutes, not most of an hour, without losing a single test. The whole suite still runs at a release and in CI, every long run has a time limit, and a new testing method is written into AGENTS.md and CLAUDE.md for every later release.

**Architecture:** The suite gets two marker tiers: `slow` (balance measurement) and `process` (starts an interpreter, gunicorn or a hosted supervisor). A bare `pytest` skips both. `--full` runs everything. Engine YAML parsing moves to libyaml through one helper, with a fallback to the pure-Python parser. The per-test temp dirs stop scanning a growing directory. pytest-xdist runs the suite in parallel. Its controller is never sandboxed, and tests are grouped by the module/class fixtures they share. pytest-testmon runs only the tests a change touched. pytest-timeout and one wrapper script (`scripts/run_tests.py`) put a limit on every run and give every run its own basetemp. CI splits its one `suite` job into three tier jobs.

```python
"""
The one way into the test suite (v0.21.1). Subcommands:

  ids      the sorted collected test ids of the FULL suite (--full --collect-only)
  fast     the default tier            (Task 2)
  full     every test                  (Task 2)
  files    explicit files/node ids     (Task 2)
  changed  only what a change touched  (Task 6)
"""

from __future__ import annotations

import argparse
import os
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent


def collected_ids(basetemp: Path) -> list[str]:
    """Every id ``pytest --full --collect-only -q`` prints, sorted."""
    done = subprocess.run(
        [sys.executable, "-m", "pytest", "tests", "--full", "--collect-only", "-q",
         "-p", "no:cacheprovider", f"--basetemp={basetemp}"],
        cwd=str(ROOT), capture_output=True, text=True, encoding="utf-8", timeout=900,
    )
    ids = [line.strip() for line in done.stdout.splitlines() if "::" in line]
    if done.returncode not in (0, 5) or not ids:
        sys.stderr.write(done.stdout[-3000:] + done.stderr[-3000:])
        raise SystemExit(f"collection failed (exit {done.returncode})")
    return sorted(ids)


def compare(before: list[str], after: list[str]) -> list[str]:
    """Lines describing every id lost or gained."""
    b, a = set(before), set(after)
    return [f"LOST  {i}" for i in sorted(b - a)] + [f"GAINED {i}" for i in sorted(a - b)]


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="run_tests.py")
    sub = parser.add_subparsers(dest="cmd", required=True)
    ids = sub.add_parser("ids")
    ids.add_argument("--out", type=Path)
    ids.add_argument("--compare", type=Path, help="a previous --out file")
    ids.add_argument("--basetemp", type=Path, required=True)
    args = parser.parse_args(argv)
    if args.cmd == "ids":
        got = collected_ids(args.basetemp)
        if args.out:
            args.out.write_text("\n".join(got) + "\n", encoding="utf-8")
        print(f"{len(got)} ids")
        if args.compare:
            diff = compare(args.compare.read_text(encoding="utf-8").split(), got)
            print("\n".join(diff) or "IDENTICAL")
            return 1 if diff else 0
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""
The suite's tiers (v0.21.1). Pure functions, so tests/test_suite_tiers.py can
test them without a pytest session.

  slow     measures balance or plays a long run. Rule 10: it KEEPS ITS LENGTH;
           only its tier moves.
  process  it, or a fixture it uses, starts a Python interpreter, gunicorn or
           a hosted supervisor (git/sh probes do not count). The conftest
           Popen wrapper enforces this marker: an unmarked test that starts
           an interpreter fails.

A BARE run (``pytest`` with no path, node id, -m or -k) leaves both tiers out.
``--full`` or CLOCKWORK_FULL_SUITE=1 keeps them. Any explicit selection runs
everything it names.
"""

from __future__ import annotations

import os
import sys
from pathlib import Path
from typing import Any

import pytest

TIER_MARKERS = ("slow", "process")
FULL_ENV = "CLOCKWORK_FULL_SUITE"
_INTERPRETERS = ("python", "pythonw", "gunicorn", "pytest")


def wants_full(config: Any) -> bool:
    return bool(config.getoption("clockwork_full", False)) or os.environ.get(FULL_ENV) == "1"


def is_bare_run(config: Any) -> bool:
    """True when the args came from ``testpaths`` and nothing narrowed them."""
    from_testpaths = config.args_source == pytest.Config.ArgsSource.TESTPATHS
    narrowed = bool(config.getoption("markexpr", "")) or bool(config.getoption("keyword", ""))
    return from_testpaths and not narrowed


def _first_word(args: Any, executable: Any) -> str:
    if executable:
        return os.fsdecode(executable)
    if isinstance(args, (str, bytes, os.PathLike)):
        text = os.fsdecode(args).strip()
        if text.startswith('"'):
            return text[1:].split('"', 1)[0]
        return text.split(" ", 1)[0]
    argv = list(args or [])
    return os.fsdecode(argv[0]) if argv else ""


def _argv(args: Any) -> list[str]:
    if isinstance(args, (str, bytes, os.PathLike)):
        return os.fsdecode(args).split()
    return [os.fsdecode(a) for a in (args or [])]


def starts_an_interpreter(args: Any, executable: Any = None) -> bool:
    """Would this Popen start Python, gunicorn or pytest?"""
    first = _first_word(args, executable)
    if not first:
        return False
    try:
        if os.path.normcase(os.path.abspath(first)) == os.path.normcase(sys.executable):
            return True
    except (OSError, ValueError):
        pass
    name = Path(first).name.lower()
    stem = name[:-4] if name.endswith(".exe") else name
    if any(stem == n or stem.startswith(n + "3") for n in _INTERPRETERS):
        return True
    argv = _argv(args)
    return any(a == "-m" and i + 1 < len(argv) and argv[i + 1] in ("gunicorn", "pytest")
               for i, a in enumerate(argv))
```

```python
"""The tier hooks (v0.21.1); see tests/tiers.py. Loaded by tests/conftest.py."""

from __future__ import annotations

from typing import Any

import pytest

from tests.tiers import TIER_MARKERS, is_bare_run, wants_full

_DESELECTED_KEY = pytest.StashKey[int]()


def pytest_addoption(parser: pytest.Parser) -> None:
    parser.addoption(
        "--full", action="store_true", dest="clockwork_full", default=False,
        help="run every tier (slow and process too). A bare `pytest` leaves them out.",
    )


def pytest_collection_modifyitems(config: pytest.Config, items: list[pytest.Item]) -> None:
    config.stash[_DESELECTED_KEY] = 0
    if wants_full(config) or not is_bare_run(config):
        return
    keep, drop = [], []
    for item in items:
        (drop if any(item.get_closest_marker(m) for m in TIER_MARKERS) else keep).append(item)
    if drop:
        config.hook.pytest_deselected(items=drop)
        items[:] = keep
    config.stash[_DESELECTED_KEY] = len(drop)


def _banner(config: pytest.Config) -> str | None:
    n = config.stash.get(_DESELECTED_KEY, 0)
    if not n:
        return None
    return f"FAST TIER: {n} slow/process tests deselected -- run --full before a release"


def pytest_report_header(config: pytest.Config) -> str | None:
    if wants_full(config):
        return "FULL SUITE: every tier"
    return "FAST TIER when bare (slow/process left out); --full for everything"


def pytest_terminal_summary(terminalreporter: Any, exitstatus: int, config: pytest.Config) -> None:
    line = _banner(config)
    if line:
        terminalreporter.write_line(line, yellow=True, bold=True)
```

```python
import datetime as _dt
import tempfile
import time

import itertools

EXIT_LIMIT = 124  # what `timeout(1)` returns: the run hit its wall-clock limit
_SEQ = itertools.count()  # two calls in one microsecond still differ

#: minutes; the bare-hang limit for a whole run (the per-test limit is pytest-timeout's)
DEFAULT_LIMITS = {"fast": 45, "full": 120, "files": 60, "changed": 45}


def basetemp_root() -> Path:
    raw = os.environ.get("CLOCKWORK_TEST_BASETEMP_ROOT")
    root = Path(raw) if raw else Path(tempfile.gettempdir()) / "claude" / "clockwork-tests"
    root = root.resolve()
    if root == ROOT or ROOT in root.parents:
        raise SystemExit(f"basetemp root {root} is inside the repository; pick one outside it")
    return root


def basetemp_for(label: str) -> Path:
    """A fresh basetemp no other run uses: pytest wipes its basetemp at start,
    so two runs sharing one destroy each other's temp trees."""
    stamp = _dt.datetime.now().strftime("%Y%m%d-%H%M%S-%f")
    return basetemp_root() / f"{label}-{os.getpid()}-{stamp}-{next(_SEQ)}"


def build_argv(args: argparse.Namespace) -> list[str]:
    argv = [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider",
            f"--basetemp={basetemp_for(args.label)}", "--durations=25"]
    if args.cmd == "full":
        argv.append("--full")
    if args.cmd == "files":
        argv += list(args.paths)
    if getattr(args, "workers", 0):
        argv += ["-n", str(args.workers)]
    return argv + list(args.extra)


def run_bounded(argv: list[str], log: Path, limit_seconds: float) -> int:
    """Run ``argv`` with its output to ``log``; past the limit, stop it
    through its own Popen handle (never a bare pid) and return EXIT_LIMIT."""
    log.parent.mkdir(parents=True, exist_ok=True)
    with open(log, "w", encoding="utf-8", errors="replace") as out:
        out.write(f"$ {' '.join(argv)}\n")
        out.flush()
        proc = subprocess.Popen(argv, cwd=str(ROOT), stdout=out, stderr=subprocess.STDOUT)
        try:
            return proc.wait(timeout=limit_seconds)
        except subprocess.TimeoutExpired:
            proc.terminate()
            try:
                proc.wait(timeout=30)
            except subprocess.TimeoutExpired:
                proc.kill()
                proc.wait(timeout=30)
            out.write(f"\nLIMIT: stopped after {limit_seconds:.0f} s (run_tests.py)\n")
            return EXIT_LIMIT
```

Extend `main` with `parse(argv)`. `fast`/`full`/`files` share `--workers N` (default 0 until Task 5), `--label` (default `run`), `--limit-minutes` (default `DEFAULT_LIMITS[cmd]`), `--keep`, `--log PATH` (default `<basetemp root>/<label>-latest.log`) and `extra` after `--`. `files` takes `paths` (nargs="+"). After the run, `main` prints the log's last 15 lines and the log path. """
One YAML loader for the engine (v0.21.1).

libyaml's CSafeLoader is ~10x faster than PyYAML's pure-Python parser
(measured 2026-10-07: HUE & CRY's 69 files, 0.73 s -> 0.07 s), and every

so the values are equal (tests/test_yamlio.py checks every shipped file). On
invalid input the pure parser runs again so the error, and its marks, read
exactly as before: validation messages and several tests quote them.

Stateless: no cache, nothing to classify in tests/fixtures/module_state.yaml.
"""

from __future__ import annotations

from typing import Any

import yaml

_FAST = getattr(yaml, "CSafeLoader", None)


def safe_load(stream_or_text: Any) -> Any:
    """``yaml.safe_load``, through libyaml when it is installed."""
    text = stream_or_text.read() if hasattr(stream_or_text, "read") else stream_or_text
    if _FAST is None:
        return yaml.safe_load(text)
    try:
        return yaml.load(text, Loader=_FAST)  # noqa: S506 -- CSafeLoader is the safe loader
    except yaml.YAMLError:
        return yaml.safe_load(text)

def xdist_group_for(item: pytest.Item) -> str | None:
    """The group an item must share a worker with, or None to spread freely.

    One skills-server port: every mcp_server test runs in one worker. Else
    the module/class/package fixtures it uses: their users stay together so
    each such fixture is built once, not once per worker. A test with none
    is free (most of the suite: ~400 of test_hue_and_cry.py's tests are).

    info = getattr(item, "_fixtureinfo", None)
    if info is None:

    names = sorted(
        name for name, defs in info.name2fixturedefs.items()
        if defs and defs[-1].scope in _WIDE

    return f"{item.nodeid.split('::', 1)[0].rsplit('/', 1)[-1]}::{'+'.join(names)}"

@dataclass
class ChangedPlan:
    mode: str                      # "none" | "testmon" | "fast"
    paths: list[str] = field(default_factory=list)
    reason: str = ""


def changed_files(base: str = "HEAD") -> list[str]:
    def git(*a: str) -> list[str]:
        out = subprocess.run(["git", *a], cwd=str(ROOT), capture_output=True, text=True,
                             timeout=60, check=True).stdout
        return [line.strip().replace("\\", "/") for line in out.splitlines() if line.strip()]
    return sorted(set(git("diff", "--name-only", base)) |
                  set(git("ls-files", "--others", "--exclude-standard")))


def plan_changed(files: list[str]) -> ChangedPlan:
    relevant = [f for f in files if not f.startswith("docs/") and not f.endswith(".md")]
    if not relevant:
        return ChangedPlan("none", reason="no code, data or test changed")
    for f in relevant:
        if f in WHOLE_TIER_FILES or (f.startswith(WHOLE_TIER_PREFIXES) and not f.endswith(".py")):
            return ChangedPlan("fast", reason=f"{f} is not Python testmon can trace")
    paths = sorted({f for f in relevant if f.startswith("tests/test_") and f.endswith(".py")} |
                   {p for prefix, ps in CHILD_CODE.items() for f in relevant
                    if f.startswith(prefix) for p in ps})
    return ChangedPlan("testmon", paths=paths, reason="Python only")
